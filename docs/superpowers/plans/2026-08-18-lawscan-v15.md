วางบล็อกที่ได้จาก Step 3 ต่อจากหัวไฟล์นี้ แล้วเติมฟังก์ชันท้ายไฟล์

```python
"""36 คอลัมน์ในลำดับของผู้ดูแล และการเขียน CSV ที่ Excel เปิดได้

ชื่อคอลัมน์คัดลอกมาตัวอักษรต่อตัวอักษร รวมช่องว่างท้ายชื่อสี่คอลัมน์ และคำว่า
``ระดับวามเสี่ยง`` ที่ตกสระอา มันดูเหมือนพิมพ์ผิดและมันไม่ใช่ของเราที่จะแก้ —
คอลัมน์ที่เปลี่ยนชื่อเงียบ ๆ จะเลิกตรงกับไฟล์อ้างอิง แล้วทุกคอลัมน์หลังจากนั้น
จะอ่านเป็นคำตอบของคำถามผิดข้อ
"""

from __future__ import annotations

import csv
import re
from pathlib import Path

NONE = "-"

# ---- วางบล็อก COLUMNS และ NONE_IS_AN_ANSWER จาก Step 3 ตรงนี้ ----

#: ชื่อกฎหมายไทยเขียนโดยเว้นวรรคหน้าคำเหล่านี้ และ text layer ของราชกิจจาฯ
#: ปิดช่องว่างนั้นเป็นประจำ
_TITLE_BREAKS = ("ว่าด้วย", "เรื่อง")

#: คลังเริ่มนับจากตรงนี้ — เอกสาร 100001 คือแถวที่ 1 ของผู้ดูแล
FIRST_DOCUMENT = 100_000


def spaced(title: str) -> str:
    for word in _TITLE_BREAKS:
        title = re.sub(rf"(?<=\S)(?<! ){re.escape(word)}", f" {word}", title)
    return title


def place_in_corpus(document: str, fallback: int) -> str:
    """ที่ของเอกสารนี้ในการนับของผู้ดูแล ไม่ใช่ตำแหน่งในรอบรัน"""
    number = (document or "").strip()
    if number.isdigit() and int(number) > FIRST_DOCUMENT:
        return str(int(number) - FIRST_DOCUMENT)
    return str(fallback)


def write(rows: list[dict[str, str]], path: Path) -> None:
    """แถวทั้งหมดลง CSV เรียงตามเลขเอกสาร

    ``utf-8-sig`` เพราะผู้ดูแลเปิดด้วย Excel และ Excel อ่านไฟล์ UTF-8 ที่ไม่มี
    BOM เป็น Latin-1 แล้วภาษาไทยทุกตัวกลายเป็นขยะ
    """
    path.parent.mkdir(parents=True, exist_ok=True)
    ordered = sorted(rows, key=lambda r: r.get("ชื่อไฟล์ ", ""))
```

```python
def row(document: str, value: dict) -> dict[str, str]:
    """คำตอบหนึ่งก้อนเป็นหนึ่งแถว

    ไม่มีการตัดสินแทนโมเดลที่นี่ มีแต่การจัดรูป — ค่าที่โมเดลตอบมาว่าอะไร
    ก็ลงช่องนั้น
    """
    from v15.answer import FIELDS

    cells: dict[str, str] = {"ชื่อไฟล์ ": document}

    for field, column in FIELDS.items():
        if field not in value:
            continue
        cell = value[field]
        if field in _DEDUPED and isinstance(cell, list):
            cell = once_each([part for v in cell for part in entries(v)])
        cells[column] = text(cell)

    cells["กฎหมายแม่"] = text(_parents(value.get("parents")))
    cells["กลุ่มเป้าหมาย"] = text(value.get("audience"))
    cells["ระดับความมั่นใจ"] = _percent(value.get("confidence"))
    return cells


#: คอลัมน์ที่โมเดลเขียนเป็นรายการแล้วพูดซ้ำตัวเอง — prompt สั่งให้ตอบชื่อกว้าง
#: คู่ชื่อเฉพาะ อ่านตรงตัวแล้วได้ชื่อกว้างหนึ่งครั้งต่อเพื่อนบ้านหนึ่งตัว
_DEDUPED = frozenset({
    "activityTags", "productGroupTags", "legalKeywordTags",
    "agencies", "licenses", "documents",
})


def _parents(parents: object) -> list[str]:
    """หนึ่งบรรทัดต่อหนึ่งมาตราที่อ้าง ซึ่งเป็นรูปที่ไฟล์อ้างอิงเขียน

    ``section`` ต่อด้วย ``item`` ไม่ใช่ f-string — strict schema ทำให้ฟิลด์ที่
    ไม่บังคับถูกประกาศเป็น nullable แล้วโมเดลตอบ ``null`` ซึ่ง f-string จะแปลง
    เป็นคำว่า ``None`` แล้วไปโผล่ในสเปรดชีต
    """
    if not isinstance(parents, list):
        return []
    lines = []
    for entry in parents:
        if not isinstance(entry, dict):
            continue
        law = item(entry.get("law"))
        if not law:
            continue
        lines.append(" ".join(filter(None, (law, item(entry.get("section"))))))
    return lines


def _percent(value: object) -> str:
    """ความมั่นใจเขียนเป็นเปอร์เซ็นต์ อย่างที่ไฟล์ของผู้ดูแลเขียน

    ตัวเลขเท่ากันทั้งสองแบบ ข้อความไม่เท่า และคอลัมน์ที่เขียนคนละรูปอ่านเป็น
    สิบช่องที่ผิดหมด
    """
    if not isinstance(value, (int, float)):
        return ""
    share = value * 100 if value <= 1 else value
    return f"{round(share)}%"
```

```python
"""ข้อความหนึ่งฉบับ และงบที่จะส่งไปให้โมเดล"""

from __future__ import annotations

from pathlib import Path

#: บอกผู้อ่านว่ามีอะไรถูกตัด — โมเดลจะได้ไม่ถูกบอกว่าเอกสารที่ถูกตัดคือเอกสารทั้งฉบับ
GAP = "\n\n[…ตัดเนื้อหาส่วนกลางออก…]\n\n"


def load(path: Path) -> str:
    return path.read_text(encoding="utf-8")


def fit(text: str, *, head: int, tail: int = 0) -> str:
    """หัวเรื่องเท่าที่งบให้ ต่อป้าย ต่อท้ายเรื่อง

    ``head=0`` แปลว่าไม่ตัด ซึ่งเป็นค่าเริ่มต้นของโปรเจกต์นี้ เอกสารที่สั้นกว่างบ
    ก็ไม่ถูกแตะอยู่แล้ว
    """
    if head <= 0 or len(text) <= head + tail:
        return text
    kept = text[:head]
    return f"{kept}{GAP}{text[-tail:]}" if tail else kept
```

```python
"""ไฟล์ที่จะสแกน จากโฟลเดอร์เดียว"""

from __future__ import annotations

from collections import defaultdict
from pathlib import Path


def gather(folder: Path) -> list[Path]:
    """ทุก ``*.txt`` ชั้นเดียว เรียงตามชื่อ ไม่ซ้ำ"""
    seen: dict[Path, None] = {}
    for path in sorted(folder.glob("*.txt")):
        seen.setdefault(path.resolve(), None)
    return list(seen)


def collisions(files: list[Path]) -> dict[str, list[Path]]:
    """ชื่อไม่รวมนามสกุลที่ซ้ำกัน — สมุดบันทึกใช้ชื่อนี้เป็นกุญแจ

    เกิดไม่ได้เมื่ออ่านโฟลเดอร์เดียว แต่ ``--text`` ชี้ไปที่อื่นได้ และคำตอบของ
    ฉบับหนึ่งทับอีกฉบับเงียบ ๆ เป็นความผิดพลาดที่ไม่มีอะไรบอก
    """
    groups: dict[str, list[Path]] = defaultdict(list)
    for path in files:
        groups[path.stem].append(path)
    return {stem: paths for stem, paths in groups.items() if len(paths) > 1}
```

```python
"""สมุดบันทึกว่าฉบับไหนถามโมเดลไปแล้ว

คำตอบบนดิสก์คือสิ่งที่การรันซ้ำใช้ต่อ วางไฟล์เพิ่มแล้วสั่งคำสั่งเดิมจึงไม่จ่ายซ้ำ
"""

from __future__ import annotations

import hashlib
import json
from pathlib import Path


def sha(path: Path) -> str:
    """แปดตัวแรกของ sha256 ของไฟล์ — พอแยกไฟล์ที่ถูก OCR ใหม่ออกจากไฟล์เดิม"""
    return hashlib.sha256(path.read_bytes()).hexdigest()[:8]


def load(answers: Path, name: str) -> dict | None:
    path = answers / f"{name}.json"
    if not path.exists():
        return None
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except ValueError:
        return None


def done(answers: Path, source: Path, by_name_only: bool = False) -> bool:
    """เคยตอบสำเร็จไว้แล้ว และข้อความยังเป็นข้อความเดิม"""
    saved = load(answers, source.stem)
    if saved is None or not saved.get("ok"):
        return False
    if by_name_only:
        return True
    recorded = saved.get("source_sha")
    return recorded is None or recorded == sha(source)


def save(answers: Path, name: str, payload: dict) -> None:
    """เขียนคำตอบไว้ — **คำตอบที่ล้มไม่ทับคำตอบที่เคยสำเร็จ**

    คีย์หมดอายุกลางทางไม่ควรทำลายงานที่จ่ายไปแล้วทั้งชั่วโมงระหว่างเดินผ่าน
    """
    answers.mkdir(parents=True, exist_ok=True)
    path = answers / f"{name}.json"
    if not payload.get("ok"):
        existing = load(answers, name)
        if existing and existing.get("ok"):
            return
    path.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
```

```python
"""prompt ที่จะส่งจริง พร้อมรายการที่มันอ้างถึง

อ่านจากดิสก์ทุกครั้ง — prompt ถูกแก้บ่อยกว่าโค้ดรอบตัวมันมาก และการต้องรีสตาร์ต
เพื่อให้เห็นการแก้ คือวิธีที่คนจะลงเอยด้วยการวัดเวอร์ชันผิด
"""

from __future__ import annotations

import json
import re
from pathlib import Path


def agencies_table(path: Path) -> str:
    """ทะเบียนหน่วยงานอย่างที่ prompt รับ — ชื่อ แท็บ กระทรวง บรรทัดละหน่วย"""
    if not path.exists():
        return ""
    register = json.loads(path.read_text(encoding="utf-8")).get("agencies", [])
    return "\n".join(
        f"{entry['name']}\t{entry.get('ministry', '')}".rstrip("\t") for entry in register
    )


def prompt(root: Path) -> str:
    """``prompts/v15.md`` พร้อมรายการที่เติมแล้ว

    placeholder ที่ไม่มีไฟล์รองรับกลายเป็นค่าว่าง ไม่ใช่ปล่อยข้อความ ``{{...}}``
    ไปให้โมเดลตีความเอง
    """
    text = (root / "prompts" / "v15.md").read_text(encoding="utf-8")
    filling = {
        "taxonomy": (root / "data" / "taxonomy.txt"),
        "agencies": None,       # มาจาก json ไม่ใช่ txt
    }
    for name in re.findall(r"\{\{(\w+)\}\}", text):
        if name == "agencies":
            value = agencies_table(root / "data" / "agencies.json")
        else:
            path = filling.get(name)
            value = path.read_text(encoding="utf-8") if path and path.exists() else ""
        text = text.replace(f"{{{{{name}}}}}", value)
    return text
```

```python
"""ที่เดียวที่คุยกับโมเดล

คำสั่งนำหน้าเสมอ เพราะ provider พวกนี้จับคู่จากตัวอักษรต้นคำขอ — เอาเอกสารขึ้นก่อน
แล้วไม่มีสองคำขอไหนแบ่ง prefix กันได้ และส่วนลดที่จ่ายค่าโปรเจกต์นี้หายไปเงียบ ๆ
"""

from __future__ import annotations

import json
import os
import time
from pathlib import Path
from typing import Any

from v15.answer import SCHEMA
from v15.read import fit

MODEL = os.environ.get("LAWSCAN_MODEL", "gpt-5-nano")

#: บันไดถอย — ชื่อขั้น งบหัวเรื่อง งบท้ายเรื่อง
#: ขั้นแรกคือเต็มฉบับตามที่ตกลง ขั้นถัดไปมีไว้ตอนล้ม ไม่ใช่ตอนยาว
RUNGS: tuple[tuple[str, int, int], ...] = (
    ("full", 0, 0),
    ("budget", 10_000, 2_500),
    ("small", 4_000, 0),
)

REQUEST_TIMEOUT = 180
RETRIES = 2


def find_key(env_file: Path) -> str:
    """คีย์ จาก environment ก่อน แล้วจากไฟล์ — ไม่ log ไม่พิมพ์ ไม่เขียนที่ไหน"""
    from_env = os.environ.get("OPENAI_API_KEY")
    if from_env:
        return from_env
    if not env_file.exists():
        return ""
    for line in env_file.read_text(encoding="utf-8").splitlines():
        name, _, value = line.strip().partition("=")
        if name.strip() == "OPENAI_API_KEY":
            return value.strip().strip("\"'")
    return ""


def ask(client: Any, instruction: str, body: str, document: str) -> dict:
    """ถามหนึ่งฉบับ ไล่บันไดลงเมื่อล้ม

    บันไดกัน "ล้ม" ได้ ไม่ได้กัน "ตอบแย่ลง" — การเรียกที่สำเร็จแต่ตอบผิดมากขึ้น
    ไม่มีอะไรที่นี่มองเห็น นั่นเป็นเรื่องของการวัด ไม่ใช่ของการเรียก
    """
    last = ""
    for name, head, tail in RUNGS:
        sent = fit(body, head=head, tail=tail)
        for attempt in range(1, RETRIES + 2):
            started = time.perf_counter()
            try:
                response = client.chat.completions.create(
                    model=MODEL,
                    messages=[
                        {"role": "system", "content": instruction},
                        {"role": "user", "content": sent},
                    ],
                    response_format={
                        "type": "json_schema",
                        "json_schema": {"name": "row", "schema": SCHEMA, "strict": True},
                    },
                )
            except Exception as exc:  # noqa: BLE001 — ฉบับเดียวล้มไม่ใช่รันล้ม
                last = f"{type(exc).__name__}: {exc}"[:300]
                break       # ขั้นนี้ไม่ไหว ลงบันได
            usage = getattr(response, "usage", None)
            text = (response.choices[0].message.content or "") if response.choices else ""
            try:
                value = json.loads(text or "{}")
            except ValueError:
                last = f"คำตอบไม่ใช่ JSON (ครั้งที่ {attempt})"
                continue
            if not value:
                last = "คำตอบว่างเปล่า"
                break
            details = getattr(usage, "prompt_tokens_details", None)
            return {
                "ok": True, "value": value, "error": "", "attempt": name,
                "input": getattr(usage, "prompt_tokens", 0) if usage else 0,
                "cached": getattr(details, "cached_tokens", 0) if details else 0,
                "output": getattr(usage, "completion_tokens", 0) if usage else 0,
                "ms": int((time.perf_counter() - started) * 1000),
            }
    return {"ok": False, "value": {}, "error": last, "attempt": "",
            "input": 0, "cached": 0, "output": 0, "ms": 0}
```

```python
def plan(text: Path, answers: Path, *, fresh: bool = False,
         by_name_only: bool = False, limit: int | None = None
         ) -> tuple[list[Path], list[Path]]:
    """ไฟล์ทั้งหมดใน text/ และไฟล์ที่ยังต้องถามโมเดล"""
    from v15 import intake, ledger

    everything = intake.gather(text)
    if fresh:
        todo = list(everything)
    else:
        todo = [p for p in everything if not ledger.done(answers, p, by_name_only)]
    if limit is not None:
        todo = todo[:limit]
    return everything, todo


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    text, out = Path(args.text), Path(args.out)
    answers = out / "answers"

    if not text.exists():
        print(f"ไม่มีโฟลเดอร์ {text} — สร้างแล้ววางไฟล์ .txt ลงไป")
        return 0

    everything, todo = plan(text, answers, fresh=args.fresh,
                            by_name_only=args.by_name_only, limit=args.limit)
    if not everything:
        print(f"ยังไม่มีไฟล์ใน {text}/ — วางไฟล์ .txt ลงไปแล้วสั่งใหม่")
        return 0

    from v15 import intake

    clash = intake.collisions(everything)
    if clash:
        for stem, paths in clash.items():
            print(f"ชื่อชนกัน {stem}: " + " · ".join(str(p) for p in paths))
        print("สมุดบันทึกใช้ชื่อไฟล์เป็นกุญแจ ชื่อซ้ำจะทับกัน — แยกชื่อก่อนแล้วสั่งใหม่")
        return 2

    changed = sum(
        1 for p in todo
        if (saved := __import__("v15.ledger", fromlist=["x"]).load(answers, p.stem))
        and saved.get("ok")
    )
    print(f"{text}/ มี {len(everything)} ไฟล์ · สแกนแล้ว {len(everything) - len(todo)}"
          f" · ต้องสแกน {len(todo)} · ข้อความเปลี่ยน {changed}")
    _print_data_dates()
    if args.dry_run:
        return 0

    if todo:
        code = _scan(todo, answers, args)
        if code:
            return code
    _write(everything, answers, out / "result.csv")
    return 0
```

`everything` ผ่าน `tidy.row` แล้วเรียก `sheet.write`
